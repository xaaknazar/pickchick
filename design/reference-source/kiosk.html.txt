      const rec = { phone, name: patch.name || (i >= 0 ? db[i].name : '') || '', updated: new Date().toISOString() };
      if (i >= 0) db[i] = { ...db[i], ...rec }; else db.push(rec);

      identifyIsPhone: s.identify === 'phone',
      phoneFg: s.phone ? 'var(--n900)' : 'var(--n600)',
      phoneMask: (() => {
        const d = s.phone.padEnd(10, '_');
        return '(' + d.slice(0, 3) + ') ' + d.slice(3, 6) + '-' + d.slice(6, 8) + '-' + d.slice(8, 10);
      })(),
      keypad: ['1', '2', '3', '4', '5', '6', '7', '8', '9', t.clear, '0', '⌫'].map(k => ({
        label: k,
        tap: () => {
          this.bump();
          if (k === '⌫') return this.setState({ phone: s.phone.slice(0, -1) });
          if (k === t.clear) return this.setState({ phone: '' });
          if (s.phone.length < 10) this.setState({ phone: s.phone + k });
        }
      })),
      phoneBtnBg: s.phone.length === 10 ? 'var(--orange)' : 'var(--n300)',
      phoneBtnFg: s.phone.length === 10 ? '#FFFFFF' : 'var(--n600)',
      confirmPhone: () => {
        if (s.phone.length !== 10) return;
        this.saveGuest({ phone: '+7' + s.phone });
        this.setState({ user: { method: 'phone' }, identify: null, idle: 0 });
        this.say('+' + this.NUM(this.chiki()) + ' ' + t.andChiki);
      },

      reviewLines: (() => {
        const rows = s.cart.map(c => ({
          name: c.name + ' ×' + c.qty, mods: c.mods || false,
          total: this.MONEY((c.price + c.extra) * c.qty), priceFg: 'var(--n900)'
        }));
        this.giftLines().forEach(g => rows.push({ name: t.gift + ': ' + g, mods: false, total: '0 ₸', priceFg: 'var(--blue)' }));
        return rows;
      })(),
      placeOrder: () => {
        if (!s.pay) return;
        const no = String(100 + Math.floor(Math.random() * 100));
        this.setState({ step: 'number', orderNo: no, doneLeft: 8, idle: 0 });
      },
      orderNo: s.orderNo || '147',
      doneDots: Array.from({ length: 8 }, (_, i) => ({ bg: i < s.doneLeft ? 'var(--orange)' : 'var(--n300)' }))
    };
  }
}

  itemTotal() {
    const it = this.current();
    if (!this.isCombo()) return (it.price + this.sizeDelta()) * this.state.itemQty;
    const d = this.DRINKS.find(x => x.id === this.state.drink);
    const s = this.SAUCES.find(x => x.id === this.state.sauce);
    const add = this.COUNTERS.reduce((a, c) => a + c.p * (this.state.counters[c.id] || 0), 0);

        const r = modRow(d, s.drink); r.pick = () => { if (!d.out) this.setState({ drink: d.id }); }; return r;
      }),
      sauceOpts: this.SAUCES.map(d => {
        const r = modRow(d, s.sauce); r.pick = () => { if (!d.out) this.setState({ sauce: d.id }); }; return r;
      }),
      drinksToggleLabel: s.expandDrinks ? 'Свернуть' : 'Показать все 18 напитков',
      toggleDrinks: () => this.setState({ expandDrinks: !s.expandDrinks }),
      counters: this.COUNTERS.map(c => ({
        name: c.name, priceLabel: '+' + this.MONEY(c.p), qty: s.counters[c.id] || 0,
        inc: () => this.bumpCounter(c.id, 1), dec: () => this.bumpCounter(c.id, -1)
      })),
      itemQty: s.itemQty,
      incItemQty: () => this.setState({ itemQty: s.itemQty + 1 }),
      decItemQty: () => this.setState({ itemQty: Math.max(1, s.itemQty - 1) }),
      itemTotalLabel: this.MONEY(this.itemTotal()),
      addToCart: () => this.addItem(), closeItem: () => this.setState({ modal: false }),

      openCart: () => this.go('cart'), closeCart: () => this.go('menu'),
      goCart: () => this.go('cart'),
      placeOrder: () => this.placeOrder(),
      showPaying: scr === 'paying', showRate: scr === 'rate',
      goTracker: () => this.go('tracker'), goMenu: () => this.go('menu'),
      goEvents: () => this.go('events'), goOrders: () => this.go('orders'), goProfile: () => this.go('profile'),
      tabMenu: tabColor('menu'), tabEvents: tabColor('events'),
      tabOrders: tabColor('orders'), tabProfile: tabColor('profile')
    };
  }
}

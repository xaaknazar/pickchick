  saveGuest(patch) {
    try {
      const key = 'pickchick.kiosk.guests';
      const db = JSON.parse(localStorage.getItem(key) || '[]');
      const phone = patch.phone || (this.state.phone ? '+7' + this.state.phone : '');
      if (!phone) return;
      const i = db.findIndex(g => g.phone === phone);
      const rec = { phone, name: patch.name || (i >= 0 ? db[i].name : '') || '', updated: new Date().toISOString() };
      if (i >= 0) db[i] = { ...db[i], ...rec }; else db.push(rec);
      localStorage.setItem(key, JSON.stringify(db));
    } catch (e) {}

    const v = document.querySelector('video[data-hero]');
    if (!v) return;
    v.muted = true; v.playsInline = true; v.loop = true;
    if (!v.__k) { v.__k = true; ['canplay', 'loadeddata', 'stalled'].forEach(e => v.addEventListener(e, () => this.kickHero())); }
    if (v.paused) { const p = v.play(); if (p && p.catch) p.catch(() => {}); }

  componentDidMount() {
    [80, 400, 1200].forEach(ms => this.timers.push(setTimeout(this.kickHero, ms)));
    this.tick = setInterval(() => {
      const s = this.state;
      if (s.step === 'attract') { this.kickHero(); return; }
      if (s.step === 'number') {
        if (s.doneLeft <= 1) this.resetAll(); else this.setState({ doneLeft: s.doneLeft - 1 });
        return;
      }
      const idleMax = (this.props.idleSeconds || 60) + 30;
      if (s.idle + 1 >= idleMax) this.resetAll(); else this.setState({ idle: s.idle + 1 });

  addLine(cart, it, mods, extra, qty) {
    const key = it.id + '|' + (mods || '');

    if (i >= 0) cart[i] = { ...cart[i], qty: cart[i].qty + qty };
    else cart.push({ key, id: it.id, name: it.name, price: it.price, extra: extra || 0, qty, mods: mods || '' });
    return cart;

        if (!w || w.step <= 2) return this.setState({ wiz: null, idle: 0 });
        this.setState({ wiz: { ...w, step: w.step - 1 }, idle: 0 });
      },
      closeWiz: () => this.setState({ wiz: null, idle: 0 }),

      upsellItems: (() => {
        const hasParty = s.cart.some(c => this.PARTY.some(p => p.id === c.id));
        const ids = hasParty ? ['d2', 'd7', 'n1', 'n2'] : ['d2', 'd5', 'd3', 'n1'];
        return ids.map(id => {
          const x = this.DOPS.find(y => y.id === id);
          const added = s.cart.some(c => c.id === id);
          return {
            name: x.name, priceLabel: this.MONEY(x.price), img: x.img || false, noImg: !x.img, slot: 'kiosk-' + x.id,
            border: added ? 'var(--blue)' : 'var(--n300)',
            btnBg: added ? 'var(--blue)' : 'var(--orange)', btnFg: '#FFFFFF', btnLabel: added ? '✓' : '+',
            add: () => {
              let cart = s.cart.slice();
              this.addLine(cart, x, '', 0, 1);
              this.setState({ cart, idle: 0 });
              this.say(x.name + ' ' + t.inOrder);
            }
          };
        });
      })(),

      cartLines: s.cart.map((c, i) => ({
        name: c.name, mods: c.mods || '—', qty: c.qty, total: this.MONEY((c.price + c.extra) * c.qty),
        slot: 'kiosk-' + c.id, img: (this.all().find(x => x.id === c.id) || {}).img || false, noImg: !(this.all().find(x => x.id === c.id) || {}).img,
        inc: () => { const n = s.cart.slice(); n[i] = { ...n[i], qty: Math.min(10, n[i].qty + 1) }; this.setState({ cart: n, idle: 0 }); },
        dec: () => { const n = s.cart.slice(); if (n[i].qty > 1) n[i] = { ...n[i], qty: n[i].qty - 1 }; else n.splice(i, 1); this.setState({ cart: n, idle: 0 }); },
        remove: () => { const n = s.cart.slice(); n.splice(i, 1); this.setState({ cart: n, idle: 0 }); }
      })),
      cartEmpty: s.cart.length === 0,
      checkoutBg: s.cart.length ? 'var(--orange)' : 'var(--n300)',
      checkoutFg: s.cart.length ? '#FFFFFF' : 'var(--n600)',
      checkoutShadow: s.cart.length ? '0 14px 32px rgba(255,103,31,.35)' : 'none',

      noUser: !s.user, hasUser: !!s.user,
      userTitle: s.user ? (s.user.method === 'qr' ? 'Азамат · Пик-мастер' : '+7 (' + s.phone.slice(0, 3) + ') ···-··-' + s.phone.slice(8)) : '',
      userBalance: s.user && s.user.method === 'qr' ? '1 240 Ч' : '',
      userAccrual: t.accrual + ': +' + this.NUM(this.chiki()) + ' ' + t.andChiki,
      userHasGifts: false,
      userPhoneOnly: !!s.user && s.user.method === 'phone',
      giftRows: [
        { id: 'g1', name: 'Фирменный соус', note: lang === 'kk' ? '5 сапар үшін' : 'за 5 визитов' },
        { id: 'g2', name: 'Картофельные дольки', note: lang === 'kk' ? 'стрик, 4 апта' : 'стрик, 4 недели' }
      ].map(g => {
        const on = !!s.gifts[g.id];
        return {
          name: g.name, note: g.note,
          check: on ? '✓' : '', boxBg: on ? 'var(--blue)' : '#FFFFFF', boxBorder: on ? 'var(--blue)' : 'var(--n300)',
          bg: on ? '#EDF3FD' : '#FFFFFF', border: on ? 'var(--blue)' : 'var(--n300)',
          toggle: () => this.setState({ gifts: { ...s.gifts, [g.id]: !on }, idle: 0 })
        };
      }),
      openIdentify: () => this.setState({ identify: 'phone', idle: 0 }),
      closeIdentify: () => this.setState({ identify: null, idle: 0 }),
      identifyTitle: t.enterPhone,
      identifyIsPhone: s.identify === 'phone',
      phoneFg: s.phone ? 'var(--n900)' : 'var(--n600)',
      phoneMask: (() => {
        const d = s.phone.padEnd(10, '_');
        return '(' + d.slice(0, 3) + ') ' + d.slice(3, 6) + '-' + d.slice(6, 8) + '-' + d.slice(8, 10);
      })(),
      keypad: ['1', '2', '3', '4', '5', '6', '7', '8', '9', t.clear, '0', '⌫'].map(k => ({
        label: k,
        tap: () => {
          this.bump();
          if (k === '⌫') return this.setState({ phone: s.phone.slice(0, -1) });
          if (k === t.clear) return this.setState({ phone: '' });
          if (s.phone.length < 10) this.setState({ phone: s.phone + k });
        }
      })),
      phoneBtnBg: s.phone.length === 10 ? 'var(--orange)' : 'var(--n300)',
      phoneBtnFg: s.phone.length === 10 ? '#FFFFFF' : 'var(--n600)',
      confirmPhone: () => {
        if (s.phone.length !== 10) return;
        this.saveGuest({ phone: '+7' + s.phone });
        this.setState({ user: { method: 'phone' }, identify: null, idle: 0 });
        this.say('+' + this.NUM(this.chiki()) + ' ' + t.andChiki);
      },

      reviewLines: (() => {
        const rows = s.cart.map(c => ({
          name: c.name + ' ×' + c.qty, mods: c.mods || false,
          total: this.MONEY((c.price + c.extra) * c.qty), priceFg: 'var(--n900)'
        }));
        this.giftLines().forEach(g => rows.push({ name: t.gift + ': ' + g, mods: false, total: '0 ₸', priceFg: 'var(--blue)' }));
        return rows;
      })(),
      placeOrder: () => {
        if (!s.pay) return;
        const no = String(100 + Math.floor(Math.random() * 100));
        this.setState({ step: 'number', orderNo: no, doneLeft: 8, idle: 0 });
      },
      orderNo: s.orderNo || '147',
      doneDots: Array.from({ length: 8 }, (_, i) => ({ bg: i < s.doneLeft ? 'var(--orange)' : 'var(--n300)' }))
    };
  }
}
